"use strict"; /* Get HTML elements */ const projectForm = document.getElementById("projectForm"); const projectName = document.getElementById("projectName"); const projectCategory = document.getElementById("projectCategory"); const projectStatus = document.getElementById("projectStatus"); const projectDate = document.getElementById("projectDate"); const projectDescription = document.getElementById("projectDescription"); const projectList = document.getElementById("projectList"); const emptyState = document.getElementById("emptyState"); const noResults = document.getElementById("noResults"); const searchInput = document.getElementById("searchInput"); const statusFilter = document.getElementById("statusFilter"); const categoryFilter = document.getElementById("categoryFilter"); const totalCount = document.getElementById("totalCount"); const progressCount = document.getElementById("progressCount"); const completedCount = document.getElementById("completedCount"); const formTitle = document.getElementById("formTitle"); const submitButton = document.getElementById("submitButton"); const cancelEditButton = document.getElementById("cancelEditButton"); const STORAGE_KEY = "studentTrackerProjects"; let projects = []; let editingProjectId = null; /* Load projects saved in this browser */ function loadProjects() { try { const saved = localStorage.getItem(STORAGE_KEY); if (saved) { const parsed = JSON.parse(saved); if (Array.isArray(parsed)) { projects = parsed.map(project => ({ ...project, category: project.category || "Other", status: project.status || "Planned", description: project.description || "", date: project.date || "" })); } } } catch (error) { console.error("Unable to load projects:", error); projects = []; } renderProjects(); } /* Save projects */ function saveProjects() { try { localStorage.setItem( STORAGE_KEY, JSON.stringify(projects) ); } catch (error) { console.error("Unable to save projects:", error); alert("Unable to save projects in this browser."); } } /* Generate a unique project ID */ function createProjectId() { if ( window.crypto && typeof window.crypto.randomUUID === "function" ) { return window.crypto.randomUUID(); } return Date.now().toString(36) + Math.random().toString(36).slice(2); } /* Add or update a project */ projectForm.addEventListener("submit", function(event) { event.preventDefault(); const name = projectName.value.trim(); if (!name) { alert("Please enter a project name."); projectName.focus(); return; } const projectData = { name: name, category: projectCategory.value, status: projectStatus.value, date: projectDate.value, description: projectDescription.value.trim() }; if (editingProjectId !== null) { const index = projects.findIndex( project => project.id === editingProjectId ); if (index !== -1) { projects[index] = { ...projects[index], ...projectData }; } } else { projects.unshift({ id: createProjectId(), ...projectData, createdAt: new Date().toISOString() }); } saveProjects(); resetForm(); renderProjects(); }); /* Reset form */ function resetForm() { projectForm.reset(); editingProjectId = null; formTitle.textContent = "Add a Project"; submitButton.textContent = "➕ Add Project"; cancelEditButton.classList.add("hidden"); } cancelEditButton.addEventListener("click", resetForm); /* Protect text before adding it to HTML */ function escapeHTML(value) { return String(value ?? "").replace( /[&<>"']/g, character => ({ "&": "&", "<": "<", ">": ">", '"': """, "'": "'" })[character] ); } /* Edit project */ function editProject(id) { const project = projects.find(item => item.id === id); if (!project) { return; } editingProjectId = id; projectName.value = project.name; projectCategory.value = project.category || "Other"; projectStatus.value = project.status || "Planned"; projectDate.value = project.date || ""; projectDescription.value = project.description || ""; formTitle.textContent = "Edit Project"; submitButton.textContent = "💾 Save Changes"; cancelEditButton.classList.remove("hidden"); projectForm.scrollIntoView({ behavior: "smooth", block: "start" }); projectName.focus(); } /* Delete project */ function deleteProject(id) { const project = projects.find(item => item.id === id); if (!project) { return; } const confirmed = confirm( 'Are you sure you want to delete "' + project.name + '"?' ); if (!confirmed) { return; } projects = projects.filter(item => item.id !== id); if (editingProjectId === id) { resetForm(); } saveProjects(); renderProjects(); } /* Format date */ function formatDate(value) { if (!value) { return "No target date"; } const parts = value.split("-"); if (parts.length !== 3) { return "No target date"; } return parts[2] + "/" + parts[1] + "/" + parts[0]; } /* Update statistics */ function updateStatistics() { totalCount.textContent = projects.length; progressCount.textContent = projects.filter( project => project.status === "In Progress" ).length; completedCount.textContent = projects.filter( project => project.status === "Completed" ).length; } /* Search and filter projects */ function getFilteredProjects() { const term = searchInput.value.trim().toLowerCase(); const selectedStatus = statusFilter.value; const selectedCategory = categoryFilter.value; return projects.filter(project => { const searchableText = [ project.name, project.category, project.status, project.description ].join(" ").toLowerCase(); const matchesSearch = searchableText.includes(term); const matchesStatus = selectedStatus === "All" || project.status === selectedStatus; const matchesCategory = selectedCategory === "All" || project.category === selectedCategory; return matchesSearch && matchesStatus && matchesCategory; }); } /* Display project cards */ function renderProjects() { updateStatistics(); const filteredProjects = getFilteredProjects(); projectList.replaceChildren(); emptyState.classList.add("hidden"); noResults.classList.add("hidden"); if (projects.length === 0) { emptyState.classList.remove("hidden"); return; } if (filteredProjects.length === 0) { noResults.classList.remove("hidden"); return; } filteredProjects.forEach(project => { const card = document.createElement("article"); card.className = "project-card"; let statusClass = "status-not-started"; if (project.status === "In Progress") { statusClass = "status-in-progress"; } else if (project.status === "Completed") { statusClass = "status-completed"; } card.innerHTML = `

📘 ${escapeHTML(project.name)}

${escapeHTML( project.description || "A new project ready to begin." )}

${escapeHTML(project.status)} ${escapeHTML(project.category)}
📅 ${escapeHTML(formatDate(project.date))}
`; card.querySelector('[data-action="edit"]') .addEventListener("click", function() { editProject(project.id); }); card.querySelector('[data-action="delete"]') .addEventListener("click", function() { deleteProject(project.id); }); projectList.appendChild(card); }); } /* Search and filter events */ searchInput.addEventListener("input", renderProjects); statusFilter.addEventListener("change", renderProjects); categoryFilter.addEventListener("change", renderProjects); /* Start the application */ loadProjects();
